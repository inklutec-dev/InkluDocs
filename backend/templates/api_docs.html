{{ _('Dokument von Anfang bis Ende mit curl') }}

# 1) anlegen
curl -X POST {{ base_url }}/api/v1/documents \
  -H "X-API-Key: idocs_deinSchluessel" \
  -F "file=@heft.pdf" -F "language=de"
# 2) Status, bis "ready"
curl {{ base_url }}/api/v1/documents/812 -H "X-API-Key: idocs_deinSchluessel"
# 3) generieren, dann Status bis "done"
curl -X POST {{ base_url }}/api/v1/documents/812/generate -H "X-API-Key: idocs_deinSchluessel"
# 4) Einträge lesen, einen korrigieren
curl {{ base_url }}/api/v1/documents/812/items -H "X-API-Key: idocs_deinSchluessel"
curl -X PATCH {{ base_url }}/api/v1/documents/812/items/5031 \
  -H "X-API-Key: idocs_deinSchluessel" -H "Content-Type: application/json" \
  -d '{"alt_text": "Zwei Kinder auf einem roten Fahrrad vor dem Rathaus"}'
# 5) exportieren
curl -X POST {{ base_url }}/api/v1/documents/812/export/pdf \
  -H "X-API-Key: idocs_deinSchluessel" -o heft_barrierefrei.pdf

{{ _('Dokument mit Python') }}

import time, requests

B = "{{ base_url }}"
H = {"X-API-Key": "idocs_deinSchluessel"}

with open("heft.pdf", "rb") as f:
    doc = requests.post(f"{B}/api/v1/documents", headers=H,
                        files={"file": f}, data={"language": "de"}).json()

def warte(status_ziel):
    while True:
        d = requests.get(f"{B}/api/v1/documents/{doc['id']}", headers=H).json()
        if d["status"] in status_ziel or d["status"] == "error":
            return d
        time.sleep(5)

warte({"ready"})
requests.post(f"{B}/api/v1/documents/{doc['id']}/generate", headers=H)
d = warte({"done"})
print(d["counts"])
items = requests.get(f"{B}/api/v1/documents/{doc['id']}/items", headers=H).json()["items"]
pdf = requests.post(f"{B}/api/v1/documents/{doc['id']}/export/pdf", headers=H)
open("heft_barrierefrei.pdf", "wb").write(pdf.content)
